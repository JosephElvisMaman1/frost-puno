# .github/workflows/backend-tests.yml

name: Backend Tests

on:
  push:
  pull_request:

jobs:
  backend-tests:
    runs-on: ubuntu-latest
    env:
      ENABLE_SUPABASE: "false"
      PYTHONPATH: ${{ github.workspace }}/backend_fastapi

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: "pip"
          cache-dependency-path: |
            backend_fastapi/requirements.txt
            backend_fastapi/requirements-dev.txt
            ml_pipeline/requirements.txt

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r backend_fastapi/requirements-dev.txt
          pip install -r ml_pipeline/requirements.txt

      - name: Run backend tests
        run: pytest backend_fastapi/tests -q

# .github/workflows/ml-training.yml

name: ML Training

on:
  workflow_dispatch:
    inputs:
      start_date:
        description: "Open-Meteo historical start date"
        required: false
        default: "2024-06-01"
      end_date:
        description: "Open-Meteo historical end date"
        required: false
        default: "2024-06-07"
      location_limit:
        description: "Number of locations for demo training. Use empty for all curated locations."
        required: false
        default: "5"
      model_version:
        description: "Model version label"
        required: false
        default: "ci-demo"
  schedule:
    - cron: "0 9 * * 1"

jobs:
  train-model:
    runs-on: ubuntu-latest
    env:
      START_DATE: ${{ github.event_name == 'workflow_dispatch' && github.event.inputs.start_date || '2024-06-01' }}
      END_DATE: ${{ github.event_name == 'workflow_dispatch' && github.event.inputs.end_date || '2024-06-07' }}
      LOCATION_LIMIT: ${{ github.event_name == 'workflow_dispatch' && github.event.inputs.location_limit || '5' }}
      MODEL_VERSION: ${{ github.event_name == 'workflow_dispatch' && github.event.inputs.model_version || 'scheduled-demo' }}

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: "pip"
          cache-dependency-path: ml_pipeline/requirements.txt

      - name: Install ML dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r ml_pipeline/requirements.txt

      - name: Ingest locations
        run: python -m ml_pipeline.data_ingestion.ingest_locations

      - name: Ingest Open-Meteo demo weather
        run: |
          if [ -n "$LOCATION_LIMIT" ]; then
            python -m ml_pipeline.data_ingestion.ingest_weather_open_meteo \
              --start-date "$START_DATE" \
              --end-date "$END_DATE" \
              --limit "$LOCATION_LIMIT" \
              --max-workers 4
          else
            python -m ml_pipeline.data_ingestion.ingest_weather_open_meteo \
              --start-date "$START_DATE" \
              --end-date "$END_DATE" \
              --max-workers 4
          fi

      - name: Build features
        run: python -m ml_pipeline.features.build_features

      - name: Validate dataset
        run: python -m ml_pipeline.preprocessing.validate_dataset

      - name: Train candidate models
        run: python -m ml_pipeline.training.train_models --version "$MODEL_VERSION"

      - name: Evaluate registered model
        run: python -m ml_pipeline.evaluation.evaluate_model

      - name: Upload model artifacts
        uses: actions/upload-artifact@v4
        with:
          name: frost-puno-model-${{ github.run_id }}
          retention-days: 14
          path: |
            ml_pipeline/registry/frost_risk_model.joblib
            ml_pipeline/registry/model_metadata.json
            ml_pipeline/evaluation/metrics_comparison.json
            ml_pipeline/evaluation/evaluation_report.json
            ml_pipeline/evaluation/confusion_matrix.csv
            data/processed/frost_training_dataset.csv



# .github/workflows/model-quality-gate.yml

name: Model Quality Gate

on:
  push:
  pull_request:
  workflow_dispatch:
    inputs:
      min_f1_macro:
        description: "Minimum accepted f1_macro"
        required: false
        default: "0.70"

jobs:
  model-quality-gate:
    runs-on: ubuntu-latest
    env:
      MIN_F1_MACRO: ${{ github.event_name == 'workflow_dispatch' && github.event.inputs.min_f1_macro || '0.70' }}

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Evaluate model metadata quality
        run: >
          python -m ml_pipeline.registry.check_model_quality
          --metadata ml_pipeline/registry/model_metadata.json

      - name: Future active-model comparison
        run: |
          echo "Future step: download the currently active production model metadata,"
          echo "compare f1_macro and drift metrics, then block deployment on regression."
